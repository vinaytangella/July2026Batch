"""
    p1 - Guess a number - hint
    computer - picks a number - hint
    computer checks the number you guessed against the picked number
    matching - congrats! it took you n times to guess correct bumber
    less - you picked a small number
    more - you picked bigger number
    keep track of guesses
"""

import random

picked_number = random.randint(1, 10)
count = 0

while True:
    guessed_number = int(input('Please enter a number between 1 and 10: '))
    count += 1  
    
    if guessed_number == picked_number:
        print(f"Congratulations! It took you {count} guesses to find the correct number.")
        break
        
    elif guessed_number > picked_number:
        print("You picked a bigger number.")
        
    else:
        print("You picked a smaller number.")
